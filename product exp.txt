    let currentScreen = "expiration-list";
    let screenHistory = [];
    let activeFilter = "All";
    let selectedTransaction = "Sold";

    function statusClass(status) {
      return status.toLowerCase().replace(/\s+/g, "-");
    }

    function statusLabel(status) {
      return status === "Nearing" ? "Nearing" : status;
    }

    function formatDate(value) {
      if (!value) return "";
      const date = new Date(value + "T00:00:00");
      return date.toLocaleDateString("en-US", {
        month: "short",
        day: "2-digit",
        year: "numeric"
      });
    }

    function showScreen(id, savePrevious = true) {
      if (savePrevious && currentScreen !== id) {
        screenHistory.push(currentScreen);
      }

      document.querySelectorAll(".screen").forEach((screen) => {
        screen.classList.remove("active");
      });

      const nextScreen = document.getElementById(id);
      if (!nextScreen) return;

      nextScreen.classList.add("active");
      currentScreen = id;

      if (id === "expiration-list") renderExpirationList();
      if (id === "batch-registration") renderBatchRegistration();
      if (id === "product-details") renderProductDetails();
      if (id === "history") renderHistory();

      window.scrollTo(0, 0);
    }

    function goBack() {
      const previous = screenHistory.pop() || "expiration-list";
      showScreen(previous, false);
    }

    function showToast(message) {
      const toast = document.getElementById("toast");
      toast.textContent = message;
      toast.style.display = "block";

      window.clearTimeout(showToast.timer);
      showToast.timer = window.setTimeout(() => {
        toast.style.display = "none";
      }, 2200);
    }

    function renderExpirationList() {
      const target = document.getElementById("expirationItems");
      const search = document.getElementById("searchInput").value.trim().toLowerCase();

      const filtered = records.filter((record) => {
        const matchesFilter =
          activeFilter === "All" || record.status === activeFilter;

        const matchesSearch =
          record.product.toLowerCase().includes(search) ||
          record.batch.toLowerCase().includes(search);

        return matchesFilter && matchesSearch;
      });

      if (!filtered.length) {
        target.innerHTML = '<div class="empty">No expiration records found.</div>';
        return;
      }

      target.innerHTML = filtered.map((record) => `
        <div class="list-item">
          <div class="row-between">
            <div>
              <p class="item-title">${record.product} / ${record.batch}</p>
              <p class="item-meta">${formatDate(record.date)} · ${record.quantity} units</p>
            </div>

            <span class="status ${statusClass(record.status)}">
              ${statusLabel(record.status)}
            </span>
          </div>

          <div class="actions" style="margin-top: 10px;">
            <button
              class="btn btn-secondary"
              type="button"
              onclick="openStock(${record.id})"
            >
              Update Stock
            </button>

            <button
              class="btn btn-secondary"
              type="button"
              onclick="openEdit(${record.id})"
            >
              Edit
            </button>
          </div>
        </div>
      `).join("");
    }

    function renderBatchRegistration() {
      const target = document.getElementById("batchRegistrationList");

      target.innerHTML = records.map((record) => `
        <div class="batch-card">
          <div class="row-between">
            <div>
              <p class="item-title">${record.batch}</p>
              <p class="item-meta">${formatDate(record.date)} · ${record.quantity} units</p>
            </div>

            <span class="status ${statusClass(record.status)}">
              ${statusLabel(record.status)}
            </span>
          </div>
        </div>
      `).join("");
    }

    function renderProductDetails() {
      const target = document.getElementById("productBatchList");
      const total = records.reduce((sum, record) => sum + Number(record.quantity), 0);

      document.getElementById("totalQuantity").textContent =
        `Total quantity: ${total} units`;

      target.innerHTML = records.map((record) => `
        <div class="batch-card">
          <div class="row-between">
            <div>
              <p class="item-title">${record.batch}</p>
              <p class="item-meta">${formatDate(record.date)}</p>
              <p class="item-meta">${record.quantity} units</p>
            </div>

            <span class="status ${statusClass(record.status)}">
              ${statusLabel(record.status)}
            </span>
          </div>
        </div>
      `).join("");
    }

    function renderHistory() {
      const target = document.getElementById("historyItems");

      target.innerHTML = historyRecords.map((record) => `
        <div class="history-item">
          <div class="history-grid">
            <div class="date">
              ${formatDate(record.date)}<br>
              ${record.time}
            </div>

            <div>
              <p class="item-title">${record.product}</p>
              <p class="item-meta">${record.batch} · ${record.quantity}</p>
            </div>

            <span class="status ${statusClass(record.status)}">
              ${record.status}
            </span>
          </div>
        </div>
      `).join("");
    }

    function openEdit(id) {
      const record = records.find((item) => item.id === id);
      if (!record) return;

      document.getElementById("editId").value = record.id;
      document.getElementById("editDate").value = record.date;
      document.getElementById("editQuantity").value = record.quantity;
      document.getElementById("editBatch").value = record.batch;
      document.getElementById("editProductLabel").textContent =
        `${record.product} · ${record.batch}`;

      const editStatus = document.getElementById("editStatus");
      editStatus.className = `status ${statusClass(record.status)}`;
      editStatus.textContent = statusLabel(record.status);

      showScreen("edit-details");
    }

    function openStock(id) {
      const record = records.find((item) => item.id === id);
      if (!record) return;

      document.getElementById("stockId").value = record.id;
      document.getElementById("stockBatchName").textContent =
        `${record.product} · ${record.batch}`;
      document.getElementById("stockAvailable").textContent =
        `${record.quantity} units remaining`;
      document.getElementById("stockQuantity").value = "";
      document.getElementById("stockNotes").value = "";

      showScreen("update-stock");
    }

    document.getElementById("addExpirationForm").addEventListener("submit", (event) => {
      event.preventDefault();

      const product = document.getElementById("product").value;
      const date = document.getElementById("expirationDate").value;
      const quantity = Number(document.getElementById("quantity").value);
      const batchInput = document.getElementById("batchNumber").value.trim();
      const error = document.getElementById("formError");

      if (!product || !date || quantity < 1) {
        error.style.display = "block";
        return;
      }

      error.style.display = "none";

      const today = new Date();
      const expiration = new Date(date + "T00:00:00");
      const days = Math.ceil((expiration - today) / 86400000);

      let status = "Normal";
      if (days < 0) status = "Expired";
      else if (days <= 30) status = "Nearing";

      const batch =
        batchInput || `LOT-${String(records.length + 1).padStart(3, "0")}`;

      records.push({
        id: Date.now(),
        product,
        batch,
        date,
        quantity,
        status
      });

      event.target.reset();
      renderBatchRegistration();
      showToast("Expiration record saved.");
      showScreen("batch-registration");
    });

    document.getElementById("editForm").addEventListener("submit", (event) => {
      event.preventDefault();

      const id = Number(document.getElementById("editId").value);
      const record = records.find((item) => item.id === id);
      if (!record) return;

      record.date = document.getElementById("editDate").value;
      record.quantity = Number(document.getElementById("editQuantity").value);
      record.batch = document.getElementById("editBatch").value.trim();

      const today = new Date();
      const expiration = new Date(record.date + "T00:00:00");
      const days = Math.ceil((expiration - today) / 86400000);

      record.status = days < 0 ? "Expired" : days <= 30 ? "Nearing" : "Normal";

      renderExpirationList();
      showToast("Expiration details updated.");
      showScreen("expiration-list");
    });

    document.getElementById("stockForm").addEventListener("submit", (event) => {
      event.preventDefault();

      const id = Number(document.getElementById("stockId").value);
      const quantity = Number(document.getElementById("stockQuantity").value);
      const notes = document.getElementById("stockNotes").value.trim();
      const record = records.find((item) => item.id === id);

      if (!record || quantity < 1) return;

      if (quantity > record.quantity) {
        showToast("Quantity exceeds the available stock.");
        return;
      }

      record.quantity -= quantity;

      historyRecords.unshift({
        date: new Date().toISOString().slice(0, 10),
        time: new Date().toLocaleTimeString("en-US", {
          hour: "2-digit",
          minute: "2-digit"
        }),
        product: record.product,
        batch: record.batch,
        quantity: `${quantity} units${notes ? " · " + notes : ""}`,
        status: selectedTransaction
      });

      renderExpirationList();
      renderHistory();
      renderProductDetails();

      showToast("Stock and transaction history updated.");
      showScreen("history");
    });

    renderExpirationList();
    renderBatchRegistration();
    renderProductDetails();
    renderHistory();
